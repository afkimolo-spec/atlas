# /etc/systemd/system/llama-completion.service
[Unit]
Description=Qwen 7B Completion Server
After=network-online.target

[Service]
Type=simple
User=llama
Group=llama
ExecStart=/opt/scripts/start-completion.sh
Restart=always
RestartSec=5
LimitNOFILE=65535
WorkingDirectory=/opt/llama.cpp
StandardOutput=append:/var/log/llama/completion.log
StandardError=append:/var/log/llama/completion.log
LimitMEMLOCK=infinity

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/llama-completion.service.d/override.conf
[Service]
LimitMEMLOCK=infinity
